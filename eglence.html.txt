        const currentQuiz = {
            id: 1,
            question: "Hangi ajanın yeteneği duvarın arkasından düşmanların yerini gösteren bir ok atar?",
            options: ["Jett", "Sova", "Cypher", "Fade"],
            answer: "Sova"
        };

        document.addEventListener("DOMContentLoaded", () => {
            checkQuizStatus();
            loadChat();
            loadLeaderboard();
        });

        function checkQuizStatus() {
            const lastQuizDate = localStorage.getItem('zerithLastQuizDate');
            const today = new Date().toDateString();

            if(lastQuizDate === today) {
                document.getElementById('questionBox').classList.add('hidden');
                const resDiv = document.getElementById('quizResult');
                resDiv.classList.remove('hidden');
                document.getElementById('resultText').innerText = "Bugünkü quiz hakkını zaten kullandın! Yarın tekrar bekleriz.";
            } else {
                renderQuiz();
            }
        }

        function renderQuiz() {
            document.getElementById('quizQuestion').innerText = currentQuiz.question;
            const optBox = document.getElementById('optionsBox');
            optBox.innerHTML = '';

            currentQuiz.options.forEach(opt => {
                optBox.innerHTML += `
                    <button onclick="answerQuiz('${opt}')" class="w-full text-left bg-[#0f1923] hover:bg-gray-800 border border-gray-700 p-2.5 rounded text-white text-sm transition">
                        ${opt}
                    </button>
                `;
            });
        }

        function answerQuiz(selected) {
            const today = new Date().toDateString();
            localStorage.setItem('zerithLastQuizDate', today);

            let message = "";
            let pointsEarned = 0;

            if(selected === currentQuiz.answer) {
                message = "Tebrikler, doğru cevap! (+100 Puan)";
                pointsEarned = 100;
            } else {
                message = "Yanlış cevap! Doğru cevap: " + currentQuiz.answer;
                pointsEarned = 0;
            }

            // Puanı liderliğe ekle (Örnek Oyuncu)
            let leaderboard = JSON.parse(localStorage.getItem('zerithLeaderboard') || '[]');
            leaderboard.push({ name: "Sen (Oyuncu)", points: pointsEarned });
            localStorage.setItem('zerithLeaderboard', JSON.stringify(leaderboard));

            document.getElementById('questionBox').classList.add('hidden');
            const resDiv = document.getElementById('quizResult');
            resDiv.classList.remove('hidden');
            document.getElementById('resultText').innerText = message;
            loadLeaderboard();
        }
